{{/* Shortcode: count-folders Counts child pages (leaf + branch bundles). Named params ONLY: section (string, optional) e.g. "galleries" or "/galleries" -> zählt alle RegularPages in dieser Section (empfohlen) path (string, optional) e.g. "/galleries" -> zählt Kinder der Zielseite (fallback) numberTag (string, optional) e.g. "code" componentTag (string, optional) e.g. "strong" Beispiele: {{< count-folders section="/galleries" >}} {{< count-folders section="galleries" numberTag="code" >}} {{< count-folders path="/galleries" componentTag="strong" >}} */}} {{- $sectionRaw := .Get "section" | default "" -}} {{- $pathRaw := .Get "path" | default "" -}} {{- $numberTag := .Get "numberTag" | default "" -}} {{- $componentTag := .Get "componentTag" | default "" -}} {{- $count := 0 -}} {{/* 1) Section-Zählung (empfohlen): alle RegularPages der Section */}} {{- if ne $sectionRaw "" -}} {{- $section := trim $sectionRaw "/" -}} {{- $count = (where site.RegularPages "Section" "==" $section) | len -}} {{/* 2) Path-Zählung (Fallback): Kinder der Zielseite zählen */}} {{- else if ne $pathRaw "" -}} {{- $path := trim $pathRaw "/" -}} {{- $target := or (site.GetPage (printf "/%s" $path)) (site.GetPage "section" $path) -}} {{- if $target -}} {{- $count = (where $target.Pages "BundleType" "ne" "") | len -}} {{- else -}} {{- $count = 0 -}} {{- end -}} {{/* 3) Kein Param: Kinder der aktuellen Seite */}} {{- else -}} {{- $count = (where .Page.Pages "BundleType" "ne" "") | len -}} {{- end -}} {{/* Dekoration */}} {{- $numOpen := cond (ne $numberTag "") (printf "<%s>" $numberTag) "" -}} {{- $numClose := cond (ne $numberTag "") (printf "" $numberTag) "" -}} {{- $compOpen := cond (ne $componentTag "") (printf "<%s>" $componentTag) "" -}} {{- $compClose := cond (ne $componentTag "") (printf "" $componentTag) "" -}} {{- printf "%s%s%d%s%s" $compOpen $numOpen $count $numClose $compClose | safeHTML -}} {{/* Zählt die Folder in der Section einer Seite (nicht rekursiv, also nicht noch deren Unterverzeichnisse) */}} {{/* {{- $section := .Get "section" | default "galleries" -}} {{ $pages := (where .Site.RegularPages "Section" "==" $section) }} {{ $pages.Len }} */}}
