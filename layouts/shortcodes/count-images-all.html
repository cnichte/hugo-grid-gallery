{{/* Shortcode: count-images-all Counts all image resources of all pages in a section (non-recursive). Named params ONLY: section (string) e.g. "galleries" or "/galleries" (leading/trailing slashes are trimmed) numberTag (string) optional, e.g. "code" componentTag (string) optional, e.g. "strong" Examples: {{< count-images-all section="galleries" >}} {{< count-images-all section="/galleries" numberTag="code" >}} {{< count-images-all section="galleries" componentTag="strong" >}} */}} {{- $numberTag := .Get "numberTag" | default "" -}} {{- $componentTag := .Get "componentTag" | default "" -}} {{- $role := .Get "role" | default "" -}} {{- $pages := partial "gallery-grid/collection-pages.html" (dict "page" .Page "role" $role) -}} {{- $count := 0 -}} {{- range $pg := $pages -}} {{- $count = add $count (len ($pg.Resources.ByType "image")) -}} {{- end -}} {{/* wraps */}} {{- $numOpen := cond (ne $numberTag "") (printf "<%s>" $numberTag) "" -}} {{- $numClose := cond (ne $numberTag "") (printf "" $numberTag) "" -}} {{- $compOpen := cond (ne $componentTag "") (printf "<%s>" $componentTag) "" -}} {{- $compClose := cond (ne $componentTag "") (printf "" $componentTag) "" -}} {{- printf "%s%s%d%s%s" $compOpen $numOpen $count $numClose $compClose | safeHTML -}}
