{{/* --------------------------------------------------------- layouts/partials/get-images.html Liefert IMMER ein Slice von Image-Resources zurück (oder leer). Akzeptiert als Input: 1) dict { "resources": []Resource } -> direkt verwenden 2) dict { "page": } -> .Resources.ByType "image" 3) (pageState / pageForShortcode / ...) -> .Resources.ByType "image" Filter: - nur lokale Dateien (haben .Name) - SVGs werden ausgeschlossen ---------------------------------------------------------- */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false -}} {{- $in := . -}} {{- $images := slice -}} {{/* 1) Explizit übergebene Resources */}} {{- if and (reflect.IsMap $in) (isset $in "resources") -}} {{- $res := (index $in "resources") -}} {{- range $res -}} {{- if and .Name (ne .MediaType.SubType "svg") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- if $debug }}{{ warnf "get-images: via .resources (%d)" (len $images) }}{{ end -}} {{- return $images -}} {{- end -}} {{/* 2) Page aus dict{page: ...} ODER direkt . */}} {{- $page := cond (and (reflect.IsMap $in) (isset $in "page")) (index $in "page") $in -}} {{ with $page }} {{ with .Resources }} {{- $res := .ByType "image" -}} {{- range $res -}} {{- if and .Name (ne .MediaType.SubType "svg") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- if $debug }}{{ warnf "get-images: from .Resources (%d) – %s" (len $images) (or $.RelPermalink $.Permalink) }}{{ end -}} {{ else }} {{- if $debug }}{{ warnf "get-images: no .Resources on %T – %s" $page (or $.RelPermalink $.Permalink) }}{{ end -}} {{ end }} {{ else }} {{- if $debug }}{{ warnf "get-images: invalid input type %T (no page/map)" $in }}{{ end -}} {{ end }} {{- return $images -}}
