{{/* --------------------------------------------------------- layouts/partials/get-images.html Liefert IMMER ein Slice von Image-Resources (lokale Dateien) oder ein leeres Slice – ohne Exceptions. Akzeptiert: 1) dict { "resources": []Resource } -> direkt verwenden 2) dict { "page": } -> .Resources.ByType "image" 3) (pageState / pageForShortcode / ...) -> .Resources.ByType "image" Filter: - nur lokale Dateien (haben .Name) - SVGs ausschließen ---------------------------------------------------------- */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false -}} {{- $in := . -}} {{- $images := slice -}} {{/* 1) Explizit übergebene Resources haben Vorrang */}} {{- if and (reflect.IsMap $in) (isset $in "resources") -}} {{- $res := (index $in "resources") -}} {{- range $res -}} {{- if and .Name (ne .MediaType.SubType "svg") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- if $debug }}{{ warnf "get-images: via .resources (%d)" (len $images) }}{{ end -}} {{- return $images -}} {{- end -}} {{/* 2) Page besorgen: dict{page: ...} oder direkt . */}} {{- $page := cond (and (reflect.IsMap $in) (isset $in "page")) (index $in "page") $in -}} {{ with $page }} {{ with .Resources }} {{- range .ByType "image" -}} {{- if and .Name (ne .MediaType.SubType "svg") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- if $debug }}{{ warnf "get-images: from .Resources (%d)" (len $images) }}{{ end -}} {{ else }} {{- if $debug }}{{ warnf "get-images: page has no .Resources" }}{{ end -}} {{ end }} {{ else }} {{- if $debug }}{{ warnf "get-images: invalid input type %T (no page/map)" $in }}{{ end -}} {{ end }} {{- return $images -}}
