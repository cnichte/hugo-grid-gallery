{{/* get-images.html Liefert IMMER ein Slice von Image-Resources zurück. Akzeptiert: - {{ partial "get-images.html" (dict "resources" $rs) }} - {{ partial "get-images.html" (dict "page" .) }} - {{ partial "get-images.html" . }} */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false -}} {{- $images := slice -}} {{/* 1) Wenn resources direkt übergeben wurden */}} {{- with .resources -}} {{- range . }} {{- if and .Name (ne .MediaType.SubType "svg") }} {{- $images = $images | append . }} {{- end }} {{- end }} {{- if $debug }}{{ warnf "get-images: resources direkt (%d)" (len $images) }}{{ end }} {{- else -}} {{/* 2) Seite bestimmen: bevorzugt .page, sonst . (aktuelles .) */}} {{- $pg := . -}} {{- with .page -}} {{- $pg = . -}} {{- end -}} {{/* 3) Aus der Seite lesen (falls Resources vorhanden) */}} {{- with $pg.Resources -}} {{- $rs := .ByType "image" -}} {{- range $rs }} {{- if and .Name (ne .MediaType.SubType "svg") }} {{- $images = $images | append . }} {{- end }} {{- end }} {{- if $debug }}{{ warnf "get-images: .Resources (%d) auf %s" (len $images) ($pg.RelPermalink | default $pg.Title) }}{{ end }} {{- else -}} {{- if $debug }}{{ warnf "get-images: keine .Resources" }}{{ end }} {{- end -}} {{- end -}} {{- return $images -}}
