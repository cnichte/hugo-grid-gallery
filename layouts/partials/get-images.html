{{/* --------------------------------------------------------- layouts/partials/get-images.html Liefert IMMER ein Slice von Image-Resources zurück (oder leer). - . kann eine Page sein - oder dict {"page": } - oder dict {"resources": <[]Resource>} ---------------------------------------------------------- */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false -}} {{- $in := . -}} {{- $images := slice -}} {{/* 1) Wenn dict{resources: ...} übergeben wurde -> direkt nehmen */}} {{- if and (reflect.IsMap $in) (isset $in "resources") -}} {{- $res := (index $in "resources") -}} {{- if $res -}} {{- range $res }} {{- if and .Name (ne .MediaType.SubType "svg") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- if $debug }}{{ warnf "get-images: .resources (%d)" (len $images) }}{{ end -}} {{- return $images -}} {{- end -}} {{/* 2) Sonst Page aus dict{page: ...} oder direkt . */}} {{- $page := cond (and (reflect.IsMap $in) (isset $in "page")) (index $in "page") $in -}} {{/* 3) Verschachtelte with-Blöcke -> kein .Resources auf nil! */}} {{ with $page }} {{ with .Resources }} {{- $res := .ByType "image" -}} {{- range $res -}} {{- if and .Name (ne .MediaType.SubType "svg") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- if $debug }}{{ warnf "get-images: .Resources (%d) für %s" (len $images) $.RelPermalink }}{{ end -}} {{ else }} {{- if $debug }}{{ warnf "get-images: Keine .Resources auf %s" $.RelPermalink }}{{ end -}} {{ end }} {{ else }} {{- if $debug }}{{ warnf "get-images: Kein valider Page-Kontext (Typ=%T)" $in }}{{ end -}} {{ end }} {{- return $images -}}
