{{/* layouts/partials/gallery-grid/create-thumbs.html */}} {{/* Erzeugt strukturierte Thumbnail-Objekte aus übergebenen Images. */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false }} {{ $images := .images | default slice }} {{ $thumbs := slice }} {{ if $debug }}{{ warnf "create-thumbs: Start processing %d images…" (len $images) }}{{ end }} {{ $re := `\d+` }} {{ if eq (len $images) 0 }} {{ if $debug }}{{ warnf "⛔ create-thumbs: Keine Bilder – abgebrochen." }}{{ end }} {{ return }} {{ end }} {{ range $index, $img := $images }} {{ with $img.RelPermalink }}{{ else }} {{ if $debug }}{{ warnf "create-thumbs: [%d] missing RelPermalink or not a resource: %v – skipped" $index $img }}{{ end }} {{ continue }} {{ end }} {{ $title := $img.Name | default "" }} {{ if eq $title "" }} {{ if $debug }}{{ warnf "create-thumbs: [%d] missing Name – skipped" $index }}{{ end }} {{ continue }} {{ end }} {{ $full := $img }} {{ $match := findRE $re $title }} {{ $number := cond (gt (len $match) 0) (index $match 0) "" }} {{/* Optional: Wasserzeichen */}} {{ with site.Data.gallery.config.watermark }} {{ $watermark := . }} {{ with resources.Get $watermark.image }} {{ $wm := . }} {{ $fw := $full.Width }} {{ $wmMaxW := div $fw 6 }} {{ if gt $wm.Width $wmMaxW }} {{ $wm = $wm.Resize (printf "%dx" $wmMaxW) }} {{ end }} {{ $posx := $watermark.posx | default "right" }} {{ $posy := $watermark.posy | default "bottom" }} {{ $x := cond (eq $posx "center") (div (sub $fw $wm.Width) 2) (cond (eq $posx "left") 20 (sub $fw (add $wm.Width 20))) }} {{ $y := cond (eq $posy "center") (div (sub $full.Height $wm.Height) 2) (cond (eq $posy "top") 20 (sub $full.Height (add $wm.Height 20))) }} {{ $full = $full.Filter (images.Overlay $wm $x $y) }} {{ end }} {{ end }} {{/* Verschiedene Thumbnails – mit Resize statt Fill für Seitenverhältnis */}} {{ $thumb20 := $img.Resize "20x" }} {{ $thumb100 := $img.Resize "100x" }} {{ $thumb250 := $img.Resize "250x" }} {{ $thumb500 := $img.Resize "500x" }} {{ $entry := dict "thumb" $thumb500.RelPermalink "full" $full.RelPermalink "width" $thumb500.Width "height" $thumb500.Height "aspectRatio" (div $thumb500.Width $thumb500.Height) "title" $title "imageTag" $number "20" $thumb20 "100" $thumb100.RelPermalink "250" $thumb250.RelPermalink "500" $thumb500.RelPermalink }} {{ if $debug }}{{ warnf "create-thumbs: [%d] created thumb entry: %q" $index $title }}{{ end }} {{ $thumbs = $thumbs | append $entry }} {{ end }} {{ if $debug }}{{ warnf "create-thumbs: Finished. %d thumbs generated." (len $thumbs) }}{{ end }} {{ return $thumbs }}
