{{/* Erzeugt strukturierte Thumbnail-Objekte aus übergebenen Images. */}} {{/* layouts/partials/gallery-grid/create-thumbs.html */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false -}} {{- $images := .images | default slice -}} {{- $config := .config | default dict -}} {{- $thumbs := slice -}} {{- if eq (len $images) 0 -}} {{- if $debug }}{{ warnf "⛔ create-thumbs: Keine Bilder – abgebrochen." }}{{ end -}} {{- return -}} {{- end -}} {{- range $index, $img := $images -}} {{- with $img.RelPermalink -}}{{- else -}} {{- if $debug }}{{ warnf "create-thumbs: [%d] missing RelPermalink – skipped" $index }}{{ end -}} {{- continue -}} {{- end -}} {{- $title := $img.Name | default "" -}} {{- if eq $title "" -}} {{- if $debug }}{{ warnf "create-thumbs: [%d] missing Name – skipped" $index }}{{ end -}} {{- continue -}} {{- end -}} {{/* Thumbs ZUERST erzeugen – VOR Wasserzeichen, damit sie clean bleiben */}} {{- $t20 := $img.Resize "20x" -}} {{- $t100 := $img.Resize "100x" -}} {{- $t250 := $img.Resize "250x" -}} {{- $t500 := $img.Resize "500x" -}} {{/* Große Version für Lightbox - längere Seite maximal maxImageSize (webtauglich) */}} {{- $maxSize := $config.maxImageSize | default 1920 -}} {{- $large := $img -}} {{- if gt $img.Width $img.Height -}} {{- /* Querformat - Breite ist länger */}} {{- if gt $img.Width $maxSize -}} {{- $large = $img.Resize (printf "%dx" $maxSize) -}} {{- end -}} {{- else -}} {{- /* Hochformat - Höhe ist länger */}} {{- if gt $img.Height $maxSize -}} {{- $large = $img.Resize (printf "x%d" $maxSize) -}} {{- end -}} {{- end -}} {{/* full ist jetzt die skalierte große Version */}} {{- $full := $large -}} {{/* Optional: Wasserzeichen NUR auf FULL anwenden. */}} {{- with $config.watermark -}} {{- $wmConf := . -}} {{- with resources.Get $wmConf.image -}} {{- $wm := . -}} {{- $fw := $full.Width -}} {{- $wmMaxW := div $fw 6 -}} {{- if gt $wm.Width $wmMaxW -}} {{- $wm = $wm.Resize (printf "%dx" $wmMaxW) -}} {{- end -}} {{- $posx := $wmConf.posx | default "right" -}} {{- $posy := $wmConf.posy | default "bottom" -}} {{- $x := cond (eq $posx "center") (div (sub $fw $wm.Width) 2) (cond (eq $posx "left") 20 (sub $fw (add $wm.Width 20))) -}} {{- $y := cond (eq $posy "center") (div (sub $full.Height $wm.Height) 2) (cond (eq $posy "top") 20 (sub $full.Height (add $wm.Height 20))) -}} {{- $full = $full.Filter (images.Overlay $wm $x $y) -}} {{- end -}} {{- end -}} {{/* Nummer aus Dateiname */}} {{- $match := findRESubmatch `-(\d{2,6})-` $title 1 -}} {{- $number := "" -}} {{- if gt (len $match) 0 -}} {{- $number = index (index $match 0) 1 -}} {{- end -}} {{- $sources := slice -}} {{- range slice $t20 $t100 $t250 $t500 -}} {{- $sources = $sources | append (dict "width" .Width "height" .Height "src" .RelPermalink) -}} {{- end -}} {{- $entry := dict "id" $img.RelPermalink "title" $title "alt" $title "width" $full.Width "height" $full.Height "src" $full.RelPermalink "sources" $sources "badge" $number -}} {{- if $debug }}{{ warnf "create-thumbs: [%d] %q ok" $index $title }}{{ end -}} {{- $thumbs = $thumbs | append $entry -}} {{- end -}} {{- if $debug }}{{ warnf "create-thumbs: Finished. %d thumbs generated." (len $thumbs) }}{{ end -}} {{- return $thumbs -}}
