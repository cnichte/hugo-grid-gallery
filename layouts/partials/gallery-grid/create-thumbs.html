{{/* Erzeugt strukturierte Thumbnail-Objekte aus übergebenen Images. */}} {{/* layouts/partials/gallery-grid/create-thumbs.html */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false -}} {{- $images := .images | default slice -}} {{- $config := .config | default dict -}} {{- $thumbs := slice -}} {{- if eq (len $images) 0 -}} {{- if $debug }}{{ warnf "⛔ create-thumbs: Keine Bilder – abgebrochen." }}{{ end -}} {{- return -}} {{- end -}} {{- $re := `\d+` -}} {{- range $index, $img := $images -}} {{- with $img.RelPermalink -}}{{- else -}} {{- if $debug }}{{ warnf "create-thumbs: [%d] missing RelPermalink – skipped" $index }}{{ end -}} {{- continue -}} {{- end -}} {{- $title := $img.Name | default "" -}} {{- if eq $title "" -}} {{- if $debug }}{{ warnf "create-thumbs: [%d] missing Name – skipped" $index }}{{ end -}} {{- continue -}} {{- end -}} {{/* Thumbs ZUERST erzeugen – VOR Wasserzeichen, damit sie clean bleiben */}} {{- $t20 := ($img.Resize "20x").RelPermalink -}} {{- $t100 := ($img.Resize "100x").RelPermalink -}} {{- $t250 := ($img.Resize "250x").RelPermalink -}} {{- $t500 := ($img.Resize "500x").RelPermalink -}} {{/* Große Version für Lightbox - längere Seite maximal maxImageSize (webtauglich) */}} {{- $maxSize := $config.maxImageSize | default 1920 -}} {{- $large := $img -}} {{- if gt $img.Width $img.Height -}} {{- /* Querformat - Breite ist länger */}} {{- if gt $img.Width $maxSize -}} {{- $large = $img.Resize (printf "%dx" $maxSize) -}} {{- end -}} {{- else -}} {{- /* Hochformat - Höhe ist länger */}} {{- if gt $img.Height $maxSize -}} {{- $large = $img.Resize (printf "x%d" $maxSize) -}} {{- end -}} {{- end -}} {{/* full ist jetzt die skalierte große Version */}} {{- $full := $large -}} {{/* Optional: Wasserzeichen NUR auf FULL anwenden. */}} {{- with $config.watermark -}} {{- $wmConf := . -}} {{- with resources.Get $wmConf.image -}} {{- $wm := . -}} {{- $fw := $full.Width -}} {{- $wmMaxW := div $fw 6 -}} {{- if gt $wm.Width $wmMaxW -}} {{- $wm = $wm.Resize (printf "%dx" $wmMaxW) -}} {{- end -}} {{- $posx := $wmConf.posx | default "right" -}} {{- $posy := $wmConf.posy | default "bottom" -}} {{- $x := cond (eq $posx "center") (div (sub $fw $wm.Width) 2) (cond (eq $posx "left") 20 (sub $fw (add $wm.Width 20))) -}} {{- $y := cond (eq $posy "center") (div (sub $full.Height $wm.Height) 2) (cond (eq $posy "top") 20 (sub $full.Height (add $wm.Height 20))) -}} {{- $full = $full.Filter (images.Overlay $wm $x $y) -}} {{- end -}} {{- end -}} {{/* Nummer aus Dateiname */}} {{- $match := findRE $re $title -}} {{- $number := cond (gt (len $match) 0) (index $match 0) "" -}} {{- /* Wenn du aspectRatio drinlassen willst, dann als Float: */ -}} {{- $ratio := div (mul 1.0 ($img.Width)) ($img.Height) -}} {{- $entry := dict "title" $title "name" $img.Name "width" $img.Width "height" $img.Height "thumb" $t500 "full" $full.RelPermalink "20" $t20 "100" $t100 "250" $t250 "500" $t500 "imageTag" $number "aspectRatio" $ratio -}} {{- if $debug }}{{ warnf "create-thumbs: [%d] %q ok" $index $title }}{{ end -}} {{- $thumbs = $thumbs | append $entry -}} {{- end -}} {{- if $debug }}{{ warnf "create-thumbs: Finished. %d thumbs generated." (len $thumbs) }}{{ end -}} {{- return $thumbs -}}
