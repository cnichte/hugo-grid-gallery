{{/* layouts/partials/gallery-grid/init.html */}} {{/* takes either a slice of .images or a .sourcePath that points to a bundle with images. An .id will be calculated if not provided. This will be used to create the URL to the data file. */}} {{/* Erzeugt strukturierte Bilddaten (JSON) aus einer Bildquelle. Funktioniert sowohl mit: – .images (Slice of Resources) – .sourcePath (z. B. "galleries/a-plague-tale") */}} {{/* layouts/partials/gallery-grid/init.html */}} {{- $debug := (index site.Params "hugo-grid-gallery").debug | default false -}} {{- $in := . -}} {{- $scratch := $in.Scratch -}} {{- $page := $in.page | default . -}} {{- $rawID := "" -}} {{- if reflect.IsMap $in -}} {{- with $in.id -}} {{- $rawID = printf "%v" . -}} {{- else -}} {{- $rawID = printf "%v" $in -}} {{- end -}} {{- else -}} {{- $rawID = printf "%v" $in -}} {{- end -}} {{- $baseID := $rawID | urlize -}} {{- if $debug }}{{ warnf "🌀 init.html called by %s – ID: %s" ($page.Title | default "") $baseID }}{{ end -}} {{/* ---------- 1) Bilderquelle robust auflösen ---------- */}} {{- $images := slice -}} {{- if reflect.IsMap $in -}} {{- if isset $in "images" -}} {{- $images = (index $in "images") | default (slice) -}} {{- else if isset $in "resources" -}} {{- $resources := (index $in "resources") | default (slice) -}} {{- range $resources -}} {{- if and .Name (ne .MediaType.SubType "svg") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- else if isset $in "page" -}} {{- $images = partial "get-images.html" (dict "page" (index $in "page")) -}} {{- else -}} {{- /* Fallback: ganze Map klappt nicht -> leer */ -}} {{- end -}} {{- else -}} {{- /* Fallback: Parameter ist eine Page */ -}} {{- $images = partial "get-images.html" (dict "page" $in) -}} {{- end -}} {{- if eq (len $images) 0 -}} {{- if $debug }}{{ warnf "🛑 init.html: Keine Bilder erhalten für ID=%s" $baseID }}{{ end -}} {{- $scratch.Set "result" (dict "imageDataUrl" "") -}} {{- return -}} {{- end -}} {{/* ---------- 2) Signatur berechnen (Cache-Busting) ---------- */}} {{- $names := slice -}} {{- range sort $images "Name" -}} {{- $names = $names | append .Name -}} {{- end -}} {{- $sig := printf "%d-%d" (len $images) (hash.FNV32a (delimit $names "|")) -}} {{- $id := printf "%s__%s" $baseID $sig | urlize -}} {{/* ---------- 3) Per-Page Scratch-Cache nutzen ---------- */}} {{- $cacheKey := printf "hgg-image-json-%s" $id -}} {{- with $scratch.Get $cacheKey -}} {{- if $debug }}{{ warnf "🧷 init.html: Cache-Hit %s → %s" $cacheKey . }}{{ end -}} {{- $scratch.Set "result" (dict "imageDataUrl" .) -}} {{- return -}} {{- end -}} {{/* ---------- 4) Thumbs erzeugen & JSON schreiben ---------- */}} {{- $thumbs := partial "gallery-grid/create-thumbs.html" (dict "images" $images) -}} {{- if $debug }}{{ warnf "🐛 init.html: %d thumbs erzeugt" (len $thumbs) }}{{ end -}} {{- $json := $thumbs | jsonify | resources.FromString (printf "%s-gallery.json" (md5 $id)) | resources.Minify -}} {{- if not $json -}} {{- if $debug }}{{ warnf "❌ init.html: Konnte JSON nicht erzeugen für ID=%s" $id }}{{ end -}} {{- $scratch.Set "result" (dict "imageDataUrl" "") -}} {{- return -}} {{- end -}} {{- $url := $json.RelPermalink -}} {{- if $debug }}{{ warnf "✅ init.html liefert an list-gallery.html = %s" $url }}{{ end -}} {{- $scratch.Set $cacheKey $url -}} {{- $scratch.Set "result" (dict "imageDataUrl" $url) -}}
